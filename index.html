function unlockSite() { const value = refs.accessInput.value.trim(); if (value === ACCESS_CODE) { accessGranted = true; document.body.classList.remove('locked'); refs.accessError.classList.remove('show'); refs.accessInput.value = ''; return; } refs.accessError.textContent = 'Incorrect access code.'; refs.accessError.classList.add('show'); refs.accessInput.focus(); } function ensureAccess() { document.body.classList.toggle('locked', !accessGranted); } refs.accessSubmit.addEventListener('click', unlockSite); refs.accessInput.addEventListener('keydown', (event) => { if (event.key === 'Enter') unlockSite(); }); ensureAccess();
